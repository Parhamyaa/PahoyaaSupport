Grundlagen

Bots erstellen

Erstelle im Discord Developer Portal eine Application, füge einen Bot hinzu, lade ihn mit den nötigen Scopes ein und lege lokal ein Node.js-Projekt an.

1. Projekt

npm init -y ausführen und anschließend npm i discord.js dotenv installieren.

2. Token

Token in eine .env-Datei legen und die Datei niemals öffentlich hochladen.

3. Start

Den Bot mit node index.js starten und in der Konsole auf eine erfolgreiche Anmeldung achten.

4. Berechtigungen

Für Moderationsbefehle braucht der Bot passende Discord-Berechtigungen und eine Rolle mit ausreichender Position.

Ticket System

Ein einfaches Slash-Command-System: /ticket erstellt einen privaten Support-Channel für den Nutzer. Ergänze später Buttons zum Schließen oder Löschen.

ticket.js
const { ChannelType, PermissionFlagsBits } = require('discord.js');

async function createTicket(interaction) {
  const channel = await interaction.guild.channels.create({
    name: `ticket-${interaction.user.username}`,
    type: ChannelType.GuildText,
    permissionOverwrites: [
      { id: interaction.guild.roles.everyone.id, deny: [PermissionFlagsBits.ViewChannel] },
      { id: interaction.user.id, allow: [PermissionFlagsBits.ViewChannel, PermissionFlagsBits.SendMessages] },
      // Hier die ID deiner Support-Rolle einsetzen:
      { id: process.env.SUPPORT_ROLE_ID, allow: [PermissionFlagsBits.ViewChannel, PermissionFlagsBits.SendMessages] }
    ]
  });

  await interaction.reply({ content: `Ticket erstellt: ${channel}`, ephemeral: true });
}

async function verify(interaction) {
  const role = interaction.guild.roles.cache.get(process.env.VERIFIED_ROLE_ID);

Moderation

Ban / Kick / Timeout

Diese Befehle sollten nur für Moderatoren verfügbar sein. Der Bot muss die jeweilige Discord-Berechtigung besitzen.

moderation.js
const { PermissionFlagsBits } = require('discord.js');

async function ban(interaction, user) {
  if (!interaction.member.permissions.has(PermissionFlagsBits.BanMembers))
    return interaction.reply({ content: 'Keine Berechtigung.', ephemeral: true });
  await interaction.guild.members.ban(user, { reason: `Ban von ${interaction.user.tag}` });
  await interaction.reply({ content: `${user.tag} wurde gebannt.` });
}

async function kick(interaction, member) {
  if (!interaction.member.permissions.has(PermissionFlagsBits.KickMembers))
    return interaction.reply({ content: 'Keine Berechtigung.', ephemeral: true });
  await member.kick(`Kick von ${interaction.user.tag}`);
  await interaction.reply({ content: `${member.user.tag} wurde gekickt.` });
}

async function timeout(interaction, member, durationMs = 10 * 60 * 1000) {
  if (!interaction.member.permissions.has(PermissionFlagsBits.ModerateMembers))
    return interaction.reply({ content: 'Keine Berechtigung.', ephemeral: true });
  await member.timeout(durationMs, `Timeout von ${interaction.user.tag}`);
  await interaction.reply({ content: `${member.user.tag} hat einen Timeout erhalten.` });
}

// Die Funktionen in deinem Slash-Command-Handler aufrufen.

Giveaways

giveaway.js
async function startGiveaway(channel, prize, durationMs) {
  const msg = await channel.send(`🎉 Giveaway: **${prize}**\nReagiere mit 🎉! Dauer: ${durationMs / 1000}s`);
  await msg.react('🎉');

  setTimeout(async () => {
    const fresh = await channel.messages.fetch(msg.id);
    const reaction = fresh.reactions.cache.get('🎉');
    const users = reaction ? await reaction.users.fetch() : new Map();
    const participants = [...users.values()].filter(u => !u.bot);

    if (!participants.length) return channel.send('Keine Teilnehmer.');
    const winner = participants[Math.floor(Math.random() * participants.length)];
    channel.send(`🎉 Gewinner: ${winner}! Preis: **${prize}**`);
  }, durationMs);
}
Entertainment

Musik-Bot

Mit /play kann der Bot dem Sprachkanal beitreten und einen angegebenen Titel bzw. eine Audio-URL abspielen. Zusätzlich gibt es Skip, Stop und Leave.

/play <titel oder URL>

Bot kommt in deinen aktuellen Sprachkanal und startet die angegebene Musik.

/skip

Überspringt den aktuellen Titel und spielt den nächsten aus der Queue.

/queue

Zeigt die Titel in der Warteschlange.

/stop

Stoppt die Wiedergabe und leert die Warteschlange.

/leave

Bot verlässt den Sprachkanal und beendet die Wiedergabe.

music.js · Play + Voice + Queue
const {
  SlashCommandBuilder,
  PermissionFlagsBits
} = require('discord.js');

const {
  joinVoiceChannel,
  createAudioPlayer,
  createAudioResource,
  AudioPlayerStatus,
  NoSubscriberBehavior,
  VoiceConnectionStatus,
  entersState
} = require('@discordjs/voice');

const play = require('play-dl');

const queues = new Map();

function getQueue(guildId) {
  if (!queues.has(guildId)) {
    const player = createAudioPlayer({
      behaviors: {
        noSubscriber: NoSubscriberBehavior.Pause
      }
    });

    const queue = {
      songs: [],
      connection: null,
      player,
      playing: false
    };

    player.on(AudioPlayerStatus.Idle, () => {
      queue.songs.shift();
      queue.playing = false;
      const textChannel = queue.textChannel;
      if (textChannel) playNext(guildId, textChannel);
    });

    queues.set(guildId, queue);
  }
  return queues.get(guildId);
}

async function playNext(guildId, channel) {
  const queue = queues.get(guildId);
  if (!queue || queue.songs.length === 0) {
    queue.playing = false;
    return;
  }

  const song = queue.songs[0];

  try {
    const stream = await play.stream(song.url, {
      discordPlayerCompatibility: true
    });

    const resource = createAudioResource(stream.stream, {
      inputType: stream.type
    });

    queue.player.play(resource);
    queue.playing = true;

    await channel.send(`🎵 Jetzt läuft: **${song.title}**`);
  } catch (error) {
    console.error('Musikfehler:', error);
    queue.songs.shift();
    await channel.send('❌ Dieser Titel konnte nicht abgespielt werden.');
    return playNext(guildId, channel);
  }
}

async function connectAndPlay(interaction, query) {
  const voiceChannel = interaction.member.voice.channel;

  if (!voiceChannel) {
    return interaction.reply({
      content: '❌ Du musst zuerst einem Sprachkanal beitreten.',
      ephemeral: true
    });
  }

  const queue = getQueue(interaction.guild.id);
  queue.textChannel = interaction.channel;

  let url = query;
  let title = query;

  // Wenn keine URL angegeben wurde, wird nach dem Titel gesucht.
  if (!/^https?:\\/\\//i.test(query)) {
    const results = await play.search(query, {
      limit: 1,
      source: { youtube: 'video' }
    });

    if (!results.length) {
      return interaction.reply({
        content: '❌ Kein passender Titel gefunden.',
        ephemeral: true
      });
    }

    url = results[0].url;
    title = results[0].title;
  }

  queue.songs.push({ title, url });

  if (!queue.connection) {
    queue.connection = joinVoiceChannel({
      channelId: voiceChannel.id,
      guildId: interaction.guild.id,
      adapterCreator: interaction.guild.voiceAdapterCreator,
      selfDeaf: true
    });

    queue.connection.subscribe(queue.player);

    try {
      await entersState(
        queue.connection,
        VoiceConnectionStatus.Ready,
        20_000
      );
    } catch (error) {
      queue.connection.destroy();
      queue.connection = null;
      return interaction.reply({
        content: '❌ Ich konnte dem Sprachkanal nicht beitreten.',
        ephemeral: true
      });
    }
  }

  if (!queue.playing) {
    await playNext(interaction.guild.id, interaction.channel);
  }

  return interaction.reply(
    `✅ **${title}** wurde ${queue.playing ? 'zur Warteschlange hinzugefügt' : 'gestartet'}.`
  );
}

async function skipMusic(interaction) {
  const queue = queues.get(interaction.guild.id);

  if (!queue || !queue.playing) {
    return interaction.reply({
      content: '❌ Aktuell läuft keine Musik.',
      ephemeral: true
    });
  }

  queue.songs.shift();
  queue.player.stop();

  return interaction.reply('⏭️ Titel übersprungen.');
}

async function stopMusic(interaction) {
  const queue = queues.get(interaction.guild.id);

  if (!queue) {
    return interaction.reply('❌ Es läuft keine Musik.');
  }

  queue.songs = [];
  queue.player.stop();
  queue.playing = false;

  if (queue.connection) {
    queue.connection.destroy();
    queue.connection = null;
  }

  return interaction.reply('⏹️ Musik gestoppt und Bot aus dem Sprachkanal entfernt.');
}

async function showQueue(interaction) {
  const queue = queues.get(interaction.guild.id);

  if (!queue || queue.songs.length === 0) {
    return interaction.reply('📭 Die Warteschlange ist leer.');
  }

  const text = queue.songs
    .slice(0, 10)
    .map((song, i) => `${i + 1}. ${song.title}`)
    .join('\\n');

  return interaction.reply(`🎶 **Warteschlange**\\n${text}`);
}

hamburg-rp.js · Kernsystem
const { Events } = require('discord.js');

const operations = new Map();
let nextOperationId = 1000;

function createOperation({ type, keyword, location, priority, caller }) {
  const id = nextOperationId++;
  const operation = {
    id,
    type,              // POLIZEI, FEUERWEHR, RD
    keyword,
    location,
    priority: priority || 'normal',
    caller,
    status: 'offen',
    units: [],
    createdAt: new Date()
  };
  operations.set(id, operation);
  return operation;
}

function assignUnit(operationId, unit) {
  const operation = operations.get(operationId);
  if (!operation) throw new Error('Einsatz nicht gefunden');
  if (!operation.units.includes(unit)) operation.units.push(unit);
  return operation;
}

function setOperationStatus(operationId, status) {
  const operation = operations.get(operationId);
  if (!operation) throw new Error('Einsatz nicht gefunden');
  operation.status = status;
  return operation;
}

function formatOperation(operation) {
  return [
    `🚨 Einsatz #${operation.id}`,
    `Art: ${operation.type}`,
    `Stichwort: ${operation.keyword}`,
    `Ort: ${operation.location}`,
    `Priorität: ${operation.priority}`,
    `Status: ${operation.status}`,
    `Einheiten: ${operation.units.join(', ') || 'keine'}`
  ].join('\\n');
}

async function reportOperation(channel, operation) {
  await channel.send(`\\`\\`\\`\\n${formatOperation(operation)}\\n\\`\\`\\``);
}

// Beispiel-Events:
// /einsatz erstellen → createOperation(...)
// /einsatz einheit → assignUnit(...)
// /einsatz status → setOperationStatus(...)
// /einsatz melden → reportOperation(...)

client.on(Events.InteractionCreate, async interaction => {
  if (!interaction.isChatInputCommand()) return;

  if (interaction.commandName === 'einsatz') {
    const sub = interaction.options.getSubcommand();

    if (sub === 'erstellen') {
      const operation = createOperation({
        type: interaction.options.getString('type', true).toUpperCase(),
        keyword: interaction.options.getString('stichwort', true),
        location: interaction.options.getString('ort', true),
        priority: interaction.options.getString('prioritaet') || 'normal',
        caller: interaction.user.id
      });

      return interaction.reply({
        content: formatOperation(operation),
        ephemeral: false
      });
    }

    if (sub === 'status') {
      const id = interaction.options.getInteger('id', true);
      const status = interaction.options.getString('status', true);
      const operation = setOperationStatus(id, status);
      return interaction.reply(formatOperation(operation));
    }
  }
});

// und für jede Funktion Discord-Rollen/Berechtigungen prüfen.
